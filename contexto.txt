"""LyricsBundle — shared result object for the lyrics pipeline.

Defines the contract between provider adapters, the cascade selector,
and the tag/file writers (Part 2).  All fields are populated by the
caller; helpers are provided for common transformations.
"""
from __future__ import annotations

import dataclasses
from typing import Optional


@dataclasses.dataclass
class LyricsBundle:
    """Normalised lyric result ready for embedding or side-car file.

    Attributes:
        kind: "word" | "line" | "plain" | "instrumental" | "none"
        lrc: Normalised LRC string with ``[mm:ss.xx]`` timestamps
             (Enhanced LRC if kind == "word").  ``None`` when kind is
             "plain", "instrumental" or "none".
        plain: Timestamp-free lyrics text, or ``None`` when kind is
               "instrumental" or "none".
        lines: List of ``(milliseconds, text)`` tuples derived from
               ``lrc``, guaranteed non-decreasing in time.
        source: Provider name plus origin identifier / URL.
        audio_duration_ms: Duration (ms) of the *final* audio file
                           (ffprobe / mutagen), used for validation.
        ref_duration_ms: Duration (ms) reported by the provider source
                         (e.g. LRCLIB track duration), used for
                         |Δduration| checks.
        offset_ms: Millisecond correction already applied to ``lines``
                   and ``lrc`` timestamps.
        confidence: Quality score 0.0 – 1.0.
        warnings: Human-readable warnings (duration mismatch, etc.).
    """

    kind: str
    lrc: Optional[str] = None
    plain: Optional[str] = None
    lines: list[tuple[int, str]] = dataclasses.field(default_factory=list)
    source: str = ""
    audio_duration_ms: Optional[int] = None
    ref_duration_ms: Optional[int] = None
    offset_ms: int = 0
    confidence: float = 0.0
    warnings: list[str] = dataclasses.field(default_factory=list)

    # ------------------------------------------------------------------
    # Derived helpers
    # ------------------------------------------------------------------

    @property
    def has_sync(self) -> bool:
        return self.kind in ("line", "word") and bool(self.lines)

    @property
    def is_plain(self) -> bool:
        return self.kind == "plain"

    @property
    def is_instrumental(self) -> bool:
        return self.kind == "instrumental"

    @property
    def is_none(self) -> bool:
        return self.kind == "none"

    def duration_delta_ms(self) -> Optional[int]:
        """|audio_duration_ms − ref_duration_ms| in ms, or None."""
        if self.audio_duration_ms is None or self.ref_duration_ms is None:
            return None
        return abs(self.audio_duration_ms - self.ref_duration_ms)

    def to_legacy_result(self, provider: str) -> dict:
        """Emit a plain dict compatible with legacy ``LyricsResult`` fields."""
        return {
            "provider": provider,
            "title": "",
            "artist": "",
            "album": "",
            "duration": (
                self.audio_duration_ms / 1000.0
                if self.audio_duration_ms is not None
                else None
            ),
            "sync_level": (
                2 if self.kind == "word" else 1 if self.kind == "line" else 0
            ),
            "plain_text": self.plain or "",
            "timed_entries": [
                (ms / 1000.0, txt) for ms, txt in self.lines
            ],
            "confidence": self.confidence,
            "warnings": list(self.warnings),
        }


def bundle_from_legacy(
    provider: str,
    title: str,
    artist: str,
    album: str,
    duration_s: Optional[float],
    sync_level: int,
    plain_text: str,
    timed_entries: list[tuple[float, str]],
    confidence: float,
    source: str = "",
    ref_duration_ms: Optional[int] = None,
    warnings: Optional[list[str]] = None,
) -> LyricsBundle:
    """Build a :class:`LyricsBundle` from legacy provider fields.

    This is a transitional helper while adapters are migrated to produce
    :class:`LyricsBundle` directly.
    """
    kind_map = {
        2: "word",
        1: "line",
        0: "plain",
    }
    kind = kind_map.get(sync_level, "plain")

    lines_ms: list[tuple[int, str]] = [
        (int(round(ts * 1000)), txt) for ts, txt in timed_entries
    ]
    lines_ms.sort(key=lambda x: x[0])

    lrc: Optional[str] = None
    if kind in ("line", "word") and lines_ms:
        lrc_lines = [
            f"[{ms // 60000:02d}:{(ms % 60000) // 1000:02d}.{(ms % 1000) // 10:02d}] {txt}"
            for ms, txt in lines_ms
        ]
        lrc = "\n".join(lrc_lines)

    return LyricsBundle(
        kind=kind,
        lrc=lrc,
        plain=plain_text or None,
        lines=lines_ms,
        source=source,
        audio_duration_ms=(
            int(round(duration_s * 1000)) if duration_s is not None else None
        ),
        ref_duration_ms=ref_duration_ms,
        confidence=max(0.0, min(1.0, confidence)),
        warnings=warnings or [],
    )

#!/usr/bin/env python3
"""Query cleaning — pure, testable module for normalising search terms.

Strips common YouTube / streaming noise from titles and artists so that
provider lookups (LRCLIB, Multi-Web) return better matches.
"""
from __future__ import annotations

import re
import unicodedata

# Tokens commonly appended to YouTube titles — removed case-insensitively.
_TITLE_NOISE = (
    r"\(official\s*(?:video|audio|lyric|visualizer|visualiser|video\s*lyrics?)\)",
    r"\[official\s*(?:video|audio|lyric|visualizer|visualiser)\]",
    r"\(lyrics?\)",
    r"\[lyrics?\]",
    r"\(lyric\s*video\)",
    r"\[lyric\s*video\]",
    r"\(hd\)",
    r"\[hd\]",
    r"\(4k\)",
    r"\[4k\]",
    r"\(remastered?\s*(?:2\d{3})?\)",
    r"\[remastered?\s*(?:2\d{3})?\]",
    r"\(live.*?\)",
    r"\[live.*?\]",
    r"\(acoustic.*?\)",
    r"\[acoustic.*?\]",
    r"\(feat\..*?\)",
    r"\[feat\..*?\]",
    r"\(ft\..*?\)",
    r"\[ft\..*?\]",
    r"\(featuring.*?\)",
    r"\[featuring.*?\]",
    r"official\s*(?:video|audio|lyric)",
    r"lyrics?\s*video",
    r"\bhd\b",
    r"\b4k\b",
)

_TITLE_NOISE_RE = re.compile("|".join(_TITLE_NOISE), re.IGNORECASE)

# Trailing " - Topic" / "Topic" suffixes on YouTube Music artist channels.
_ARTIST_TOPIC_RE = re.compile(r"\s*-\s*Topic\s*$", re.IGNORECASE)

# Multiple whitespace.
_WS_RE = re.compile(r"\s+")


def normalize_unicode(value: str) -> str:
    """NFKD-normalise, strip combining marks, lowercase."""
    if not value:
        return ""
    decomposed = unicodedata.normalize("NFKD", value)
    stripped = "".join(
        ch for ch in decomposed if not unicodedata.combining(ch)
    )
    return stripped.casefold().strip()


def clean_title(title: str) -> str:
    """Remove YouTube noise from a track title.

    Examples:
        "Bohemian Rhapsody (Official Video)" → "Bohemian Rhapsody"
        "Shape of You [Lyrics]" → "Shape of You"
        "Blinding Lights (feat. Kendrick Lamar)" → "Blinding Lights"
    """
    if not title:
        return title
    cleaned = _TITLE_NOISE_RE.sub("", title)
    cleaned = cleaned.replace("–", "-").replace("—", "-")
    # Normalise dash separators but keep meaningful ones inside title
    cleaned = _WS_RE.sub(" ", cleaned).strip(" -–—|")
    return cleaned.strip()


def clean_artist(artist: str) -> str:
    """Remove YouTube Music " - Topic" channel suffix and noise."""
    if not artist:
        return artist
    cleaned = _ARTIST_TOPIC_RE.sub("", artist)
    cleaned = _WS_RE.sub(" ", cleaned).strip(" -–—|")
    return cleaned.strip()


def split_feat(title: str) -> tuple[str, str | None]:
    """Split ``feat.`` / ``ft.`` artists out of a title.

    Returns ``(clean_title, feat_artist)``.
    """
    if not title:
        return title, None
    m = re.search(
        r"\s+(?:feat\.?|ft\.?|featuring)\s+(.+?)\s*$", title, re.IGNORECASE
    )
    if m:
        return title[: m.start()].strip(), m.group(1).strip()
    return title, None


def build_query_terms(
    title: str,
    artist: str,
    album: str = "",
    feat_artist: str | None = None,
) -> dict[str, str]:
    """Build cleaned query terms for provider lookups.

    Returns a dict with keys ``title``, ``artist``, ``album`` and
    ``search`` (combined title + artist for fallback providers).
    """
    clean_t = clean_title(title)
    clean_a = clean_artist(artist)

    # Move feat. artists from title into artist field for better matching
    if feat_artist:
        if clean_a:
            clean_a = f"{clean_a} {feat_artist}"
        else:
            clean_a = feat_artist

    return {
        "title": clean_t,
        "artist": clean_a,
        "album": clean_title(album) if album else "",
        "search": f"{clean_t} {clean_a}".strip(),
    }

#!/usr/bin/env python3
"""LRC validation & normalisation — pure, testable module.

Accepts ``[mm:ss]``, ``[mm:ss.x]``, ``[mm:ss.xx]`` and ``[mm:ss.xxx]``.
Expands multi-timestamp lines, sorts, enforces monotonicity, strips
advertising / credit lines, preserves ``[ar:]``/``[ti:]`` headers, and
normalises every timestamp to ``[mm:ss.xx]``.
"""
from __future__ import annotations

import re
from dataclasses import dataclass

# Header lines kept verbatim (not treated as timed entries).
_HEADER_RE = re.compile(
    r"^\[(?:ti|ar|al|au|by|offset|length|re|ve)\:[^\]]*\]\s*$",
    re.IGNORECASE,
)

# Timed line: [mm:ss] or [mm:ss.x] / [mm:ss.xx] / [mm:ss.xxx]  text
_TIMED_RE = re.compile(
    r"^\[(\d{1,2}):(\d{2})(?:\.(\d{1,3}))?\]\s*(.*)$"
)

# Any timestamp tag anywhere on the line (for multi-tag expansion).
_ALL_TAGS_RE = re.compile(
    r"\[(\d{1,2}):(\d{2})(?:\.(\d{1,3}))?\]"
)

# Advertising / credit patterns — matched case-insensitively.
_AD_PATTERNS = (
    r"letra\s*y\s*música",
    r"lyrics?\s*by",
    r"©\s*\d{4}",
    r"all rights reserved",
    r"subscríbete",
    r"subscribe",
    r"publicidad",
    r"advertisement",
    r"descarga.*(?:música|mp3)",
    r"www\.",
    r"http://",
    r"https://",
)

_AD_RE = re.compile("|".join(_AD_PATTERNS), re.IGNORECASE)

# Minimum fraction of non-empty lines that must carry a timestamp.
_MIN_TIMESTAMP_RATIO = 0.60

# Tolerance: last timestamp may exceed audio duration by this many seconds.
_LAST_TS_TOLERANCE_S = 3.0


@dataclass
class LrcValidationResult:
    """Outcome of :func:`validate_and_normalise_lrc`."""

    valid: bool
    entries: list[tuple[int, str]]  # (ms, text), sorted, monotonic
    headers: dict[str, str]         # [ti:], [ar:] etc.
    plain: str                      # timestamp-free text
    reason: str = ""                # human-readable failure reason
    warnings: list[str] = None      # type: ignore[assignment]

    def __post_init__(self) -> None:
        if self.warnings is None:
            self.warnings = []


def _ms_from_tags(mins: int, secs: int, ms_part: str | None) -> int:
    """Convert regex capture groups to total milliseconds."""
    total = mins * 60_000 + secs * 1000
    if ms_part:
        # Pad / truncate to 3 digits (ms).
        ms_part = (ms_part + "000")[:3]
        total += int(ms_part)
    return total


def _format_ts(ms: int) -> str:
    """Format milliseconds as ``[mm:ss.xx]``."""
    m, s = divmod(ms // 1000, 60)
    ms_part = ms % 1000
    return f"[{m:02d}:{s:02d}.{ms_part // 10:02d}]"


def _looks_like_ad(line: str) -> bool:
    """Heuristic: does the line look like advertising or credits?"""
    stripped = line.strip()
    if not stripped:
        return False
    # Very short lines (1-2 chars) are usually not lyrics.
    if len(stripped) <= 2 and not stripped.isalnum():
        return True
    return bool(_AD_RE.search(stripped))


def parse_lrc(text: str) -> list[tuple[int, str]]:
    """Parse LRC text into ``[(ms, text), ...]`` entries.

    Handles:
        * Single timestamp per line: ``[mm:ss.xx] text``
        * Multiple timestamps per line: ``[00:01.00][00:01.50] text``
        * Headers: ``[ti: ...]`` etc. (ignored here, see :func:`split_headers`)
        * 1–3 digit fractional seconds (padded/truncated to ms).
    """
    entries: list[tuple[int, str]] = []
    for raw_line in text.splitlines():
        line = raw_line.strip()
        if not line or _HEADER_RE.match(line):
            continue
        # Multi-tag expansion: find every timestamp tag on the line.
        tags = _ALL_TAGS_RE.findall(line)
        if not tags:
            continue
        # Text is everything after the last timestamp tag.
        text_part = _ALL_TAGS_RE.sub("", line, count=len(tags)).strip()
        for mins_s, secs_s, ms_s in tags:
            ms = _ms_from_tags(int(mins_s), int(secs_s), ms_s)
            entries.append((ms, text_part))
    return entries


def split_headers(text: str) -> tuple[dict[str, str], str]:
    """Return ``(headers, body)`` where headers contains ``[ti:]`` etc."""
    headers: dict[str, str] = {}
    body_lines: list[str] = []
    for raw_line in text.splitlines():
        line = raw_line.strip()
        m = re.match(r"^\[(\w+)\:\s*(.*)\]\s*$", line, re.IGNORECASE)
        if m:
            headers[m.group(1).lower()] = m.group(2).strip()
        else:
            body_lines.append(raw_line)
    return headers, "\n".join(body_lines)


def validate_and_normalise_lrc(
    lrc_text: str,
    audio_duration_s: float | None = None,
) -> LrcValidationResult:
    """Validate and normalise an LRC string.

    Returns a :class:`LrcValidationResult` with sorted, monotonic entries
    in millisecond precision, headers extracted, and plain-text body.

    Rules:
        * At least 60 % of non-empty lines must have a timestamp.
        * Timestamps must be non-decreasing.
        * Last timestamp ≤ audio duration + 3 s tolerance (if duration given).
        * Advertising / credit lines are removed.
    """
    warnings: list[str] = []
    headers, body = split_headers(lrc_text)

    entries = parse_lrc(body)

    # Count non-empty lines for the 60 % ratio.
    non_empty_lines = [
        ln for ln in body.splitlines() if ln.strip()
    ]
    if non_empty_lines:
        ratio = len(entries) / len(non_empty_lines)
        if ratio < _MIN_TIMESTAMP_RATIO:
            return LrcValidationResult(
                valid=False,
                entries=[],
                headers=headers,
                plain=body.strip(),
                reason=(
                    f"solo {ratio:.0%} de líneas con timestamp "
                    f"(mínimo {_MIN_TIMESTAMP_RATIO:.0%})"
                ),
                warnings=warnings,
            )

    # Remove ad lines.
    filtered: list[tuple[int, str]] = []
    for ms, txt in entries:
        if _looks_like_ad(txt):
            warnings.append(f"línea publicitaria eliminada: {txt[:40]}...")
            continue
        if txt.strip():
            filtered.append((ms, txt))

    if not filtered:
        return LrcValidationResult(
            valid=False,
            entries=[],
            headers=headers,
            plain=body.strip(),
            reason="no hay líneas de letra válidas tras filtrar",
            warnings=warnings,
        )

    # Sort by time.
    filtered.sort(key=lambda x: x[0])

    # Enforce non-decreasing timestamps (stable sort keeps original order
    # for equal timestamps).
    monotonic: list[tuple[int, str]] = []
    last_ms = -1
    for ms, txt in filtered:
        if ms < last_ms:
            warnings.append(
                f"timestamp no monotónico corregido: {ms}ms < {last_ms}ms"
            )
            ms = last_ms
        monotonic.append((ms, txt))
        last_ms = ms

    # Duration sanity check.
    if audio_duration_s is not None:
        last_ts_s = monotonic[-1][0] / 1000.0
        if last_ts_s > audio_duration_s + _LAST_TS_TOLERANCE_S:
            return LrcValidationResult(
                valid=False,
                entries=monotonic,
                headers=headers,
                plain="\n".join(t for _, t in monotonic),
                reason=(
                    f"último timestamp ({last_ts_s:.1f}s) excede "
                    f"duración de audio ({audio_duration_s:.1f}s) + "
                    f"{_LAST_TS_TOLERANCE_S}s"
                ),
                warnings=warnings,
            )

    # Build plain text.
    plain = "\n".join(txt for _, txt in monotonic)

    return LrcValidationResult(
        valid=True,
        entries=monotonic,
        headers=headers,
        plain=plain,
        warnings=warnings,
    )


def normalise_lrc(entries: list[tuple[int, str]]) -> str:
    """Render ``[(ms, text), ...]`` as an LRC string with ``[mm:ss.xx]``."""
    lines = [_format_ts(ms) + (" " + txt if txt else "") for ms, txt in entries]
    return "\n".join(lines)


def shift_lrc(lrc_text: str, ms_offset: int) -> str:
    """Apply a millisecond offset to every timestamp in an LRC string.

    Positive ``ms_offset`` shifts later; negative shifts earlier.
    Timestamps are clamped at 0 ms.  Sign convention:
        * ``shift_lrc(lrc, +500)`` → every line appears 0.5 s later.
        * ``shift_lrc(lrc, -500)`` → every line appears 0.5 s earlier.
    """
    entries = parse_lrc(lrc_text)
    shifted = [(max(0, ms + ms_offset), txt) for ms, txt in entries]
    shifted.sort(key=lambda x: x[0])
    return normalise_lrc(shifted)

#!/usr/bin/env python3
"""Local query cache for lyric lookups.

Key: artist + title + duration.  Value: serialised LyricsBundle dict.
Persisted as JSON under a cache directory so repeated lookups for the same
track do not hit the network.
"""
from __future__ import annotations

import hashlib
import json
import os
from pathlib import Path
from typing import Optional


def _cache_key(artist: str, title: str, duration_ms: Optional[int]) -> str:
    """Stable SHA-256 key for a query."""
    parts = [
        artist.casefold().strip(),
        title.casefold().strip(),
        str(duration_ms) if duration_ms is not None else "",
    ]
    return hashlib.sha256("|".join(parts).encode("utf-8")).hexdigest()


class LyricsCache:
    """Simple filesystem cache keyed by (artist, title, duration)."""

    def __init__(self, cache_dir: str | Path, ttl_seconds: int = 604800):
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(parents=True, exist_ok=True)
        self.ttl_seconds = ttl_seconds

    def _path(self, key: str) -> Path:
        return self.cache_dir / f"{key}.json"

    def get(
        self, artist: str, title: str, duration_ms: Optional[int] = None
    ) -> Optional[dict]:
        """Return cached bundle dict or None if missing / expired."""
        key = _cache_key(artist, title, duration_ms)
        p = self._path(key)
        if not p.is_file():
            return None
        try:
            data = json.loads(p.read_text(encoding="utf-8"))
        except (json.JSONDecodeError, OSError):
            return None
        ts = data.get("cached_at", 0)
        if ts and (os.times().elapsed - ts) > self.ttl_seconds:
            try:
                p.unlink()
            except OSError:
                pass
            return None
        return data.get("bundle")

    def set(
        self,
        artist: str,
        title: str,
        bundle: dict,
        duration_ms: Optional[int] = None,
    ) -> None:
        """Serialise and store a bundle dict."""
        key = _cache_key(artist, title, duration_ms)
        p = self._path(key)
        payload = {
            "cached_at": os.times().elapsed,
            "bundle": bundle,
        }
        try:
            p.write_text(json.dumps(payload, ensure_ascii=False), encoding="utf-8")
        except OSError:
            pass  # Cache failure is non-fatal.

    def clear(self) -> int:
        """Remove all cache files.  Returns number of files deleted."""
        count = 0
        for p in self.cache_dir.glob("*.json"):
            try:
                p.unlink()
                count += 1
            except OSError:
                pass
        return count

"""Offset / shift helpers for LRC timestamps.

Sign convention (documented here and in :func:`shift_lrc`):

    * Positive offset  → lyrics appear *later*  (add ms to each timestamp)
    * Negative offset  → lyrics appear *earlier* (subtract ms)

and the header is not relied upon.

    lrc_text: str,
    ms_offset: int,

    Returns a normalised LRC string.  Wraps :func:`shift_lrc` for
    clarity; see that function for sign convention.
    """
    return _shift_lrc_text(lrc_text, ms_offset)

    current_ts_ms: int,
    anchor_ts_ms: int,
) -> int:
    """Return the ms needed to move ``current_ts_ms`` to ``anchor_ts_ms``.

    ``offset = anchor - current``.  Positive = shift later.
    """

    results: list[LyricsResult],

def _normalize_match_text(value: str) -> str:
    decomposed = unicodedata.normalize("NFKD", value).casefold()
    return "".join(
        ch
        for ch in decomposed
        if not unicodedata.combining(ch) and ch.isalnum()
    )


def _bundle_matches_query(bundle: LyricsBundle, query: SongQuery) -> bool:
    """Check title/artist compatibility between bundle and query."""
    if query.isrc and bundle.isrc:
        return bundle.isrc.casefold() == query.isrc.casefold()
    for actual, expected in (
        (bundle.title, query.title),
        (bundle.artist, query.artist),
    ):
        if actual and expected:
            if _normalize_match_text(actual) != _normalize_match_text(expected):
                return False
    return True

def split_feat(title: str) -> tuple[str, str | None]:
    """Split ``feat.`` / ``ft.`` artists out of a title.

    Handles ``(feat. ...)`` and ``(ft. ...)`` parenthetical forms.
    Returns ``(clean_title, feat_artist)``.
    """
    if not title:
        return title, None
    # Try parenthetical form first.
    m = _FEAT_PAREN_RE.search(title)
    if m:
        inner = m.group(0)
        feat_match = re.search(
            r"\b(?:feat\.?|ft\.?|featuring)\s+(.+?)\s*\)", inner, re.IGNORECASE
        )
        if feat_match:
            return title[: m.start()].strip(), feat_match.group(1).strip()
    # Fallback to standard form.
    m = re.search(
        r"\s+(?:feat\.?|ft\.?|featuring)\s+(.+?)\s*$", title, re.IGNORECASE
    )
